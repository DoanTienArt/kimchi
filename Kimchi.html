<script>

  let activeBox = null;

  let startY = 0;

  let startVal = 0;


  const boxes = document.querySelectorAll('.scrub-box');


  function formatNumber(val, decimals) {

    return decimals > 0 ? val.toFixed(decimals) : Math.round(val).toString();

  }


  // Tự động cập nhật các dòng con khi kéo dòng Cải thảo gốc

  function updateAllByRatio(newMasterVal) {

    boxes.forEach(box => {

      if (box.dataset.master === "true") return;

      const base = parseFloat(box.dataset.base);

      const decimals = parseInt(box.dataset.decimals || "0");

      const calculated = base * newMasterVal;

      box.textContent = formatNumber(calculated, decimals);

    });

  }


  // Bắt đầu kéo (Chuột & Cảm ứng điện thoại)

  function startDrag(e, box) {

    activeBox = box;

    startY = e.touches ? e.touches[0].clientY : e.clientY;

    startVal = parseFloat(box.textContent);

    box.classList.add('active');

    document.body.style.cursor = 'ns-resize';

  }


  // Đang kéo lên/xuống

  function onDrag(e) {

    if (!activeBox) return;

    if (e.cancelable) e.preventDefault();


    const currentY = e.touches ? e.touches[0].clientY : e.clientY;

    const step = parseFloat(activeBox.dataset.step) || 1;

    const min = parseFloat(activeBox.dataset.min) ?? 0;

    const decimals = parseInt(activeBox.dataset.decimals || "0");


    // Kéo lên (startY - currentY > 0) -> tăng số; kéo xuống -> giảm số

    const deltaY = startY - currentY;

    const stepsMoved = Math.round(deltaY / 6); // 6px = 1 bước nhảy

    let newVal = Math.max(min, startVal + stepsMoved * step);


    activeBox.textContent = formatNumber(newVal, decimals);


    if (activeBox.dataset.master === "true") {

      updateAllByRatio(newVal);

    }

  }


  // Kết thúc kéo

  function endDrag() {

    if (!activeBox) return;

    activeBox.classList.remove('active');

    activeBox = null;

    document.body.style.cursor = 'default';

  }


  boxes.forEach(box => {

    // Sự kiện chuột

    box.addEventListener('mousedown', (e) => startDrag(e, box));


    // Sự kiện cảm ứng (Mobile/Tablet)

    box.addEventListener('touchstart', (e) => startDrag(e, box), { passive: false });


    // Hỗ trợ lăn bánh xe chuột trực tiếp trên con số

    box.addEventListener('wheel', (e) => {

      e.preventDefault();

      const step = parseFloat(box.dataset.step) || 1;

      const min = parseFloat(box.dataset.min) ?? 0;

      const decimals = parseInt(box.dataset.decimals || "0");

      let currentVal = parseFloat(box.textContent);

      

      currentVal += (e.deltaY < 0 ? step : -step);

      currentVal = Math.max(min, currentVal);

      box.textContent = formatNumber(currentVal, decimals);


      if (box.dataset.master === "true") {

        updateAllByRatio(currentVal);

      }

    }, { passive: false });

  });


  window.addEventListener('mousemove', onDrag);

  window.addEventListener('touchmove', onDrag, { passive: false });

  window.addEventListener('mouseup', endDrag);

  window.addEventListener('touchend', endDrag);


  function resetDefaults() {

    boxes.forEach(box => {

      const base = parseFloat(box.dataset.base);

      const decimals = parseInt(box.dataset.decimals || "0");

      box.textContent = formatNumber(base, decimals);

    });

  }

</script>
